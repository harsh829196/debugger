from mcp.server.fastmcp import FastMCP
import os, subprocess
from dotenv import load_dotenv

load_dotenv()

mcp = FastMCP("coding-debugger")

SKIP = {".git", "venv", ".venv", "__pycache__", "node_modules", ".idea", "dist", "build"}
MAX_CHARS = 6000

# Tools can only touch files under this folder.
# Defaults to the folder you launch agent.py from; override with DEBUG_ROOT in .env.
ROOT = os.path.realpath(os.getenv("DEBUG_ROOT", "."))


def clip(text: str) -> str:
    if len(text) > MAX_CHARS:
        return text[:MAX_CHARS] + f"\n...[truncated, {len(text)} chars total]"
    return text


def safe_path(path: str) -> str:
    """Resolve a path and make sure it stays inside ROOT."""
    full = os.path.realpath(os.path.join(ROOT, path))
    if os.path.commonpath([full, ROOT]) != ROOT:
        raise ValueError(f"Path '{path}' is outside the allowed root.")
    return full


@mcp.tool()
def list_files(directory: str = ".", max_results: int = 150) -> str:
    """List files in a directory recursively, skipping junk folders."""
    try:
        base = safe_path(directory)
        results = []
        for root, dirs, files in os.walk(base):
            dirs[:] = [d for d in dirs if d not in SKIP]
            for f in files:
                results.append(os.path.relpath(os.path.join(root, f), ROOT))
                if len(results) >= max_results:
                    return clip("\n".join(results) + f"\n...truncated at {max_results} files")
        return clip("\n".join(results) or "No files found.")
    except Exception as e:
        return f"Error listing files: {e}"


@mcp.tool()
def read_file(path: str, start_line: int = 1, end_line: int = 150) -> str:
    """Read a file between line numbers (max 150 lines per call)."""
    try:
        full = safe_path(path)
        start_line = max(1, start_line)
        end_line = min(end_line, start_line + 149)
        with open(full, "r", errors="replace") as f:
            lines = f.readlines()
        chunk = "".join(lines[start_line - 1:end_line])
        return clip(chunk or f"No content in lines {start_line}-{end_line} (file has {len(lines)} lines).")
    except Exception as e:
        return f"Error reading {path}: {e}"


@mcp.tool()
def search_code(pattern: str, directory: str = ".") -> str:
    """Search code using ripgrep (max 40 matching lines)."""
    try:
        base = safe_path(directory)
        result = subprocess.run(
            ["rg", "-n", "--max-count", "5", "--max-columns", "200",
             "-g", "!.venv", "-g", "!venv", "-g", "!node_modules",
             "-g", "*.py", "-g", "*.md", "-g", "*.txt", "-g", "*.yaml", "-g", "*.yml",
             pattern, base],
            capture_output=True, text=True, timeout=15,
        )
        out = result.stdout or result.stderr or "No matches found."
        return clip("\n".join(out.splitlines()[:40]))
    except Exception as e:
        return f"Error running search: {e}"

import shlex

PROTECTED_BRANCHES = {"main", "master"}


def git(*args: str) -> str:
    r = subprocess.run(["git", *args], cwd=ROOT, capture_output=True, text=True, timeout=60)
    return (r.stdout + r.stderr).strip() or "(no output)"


def current_branch() -> str:
    return git("rev-parse", "--abbrev-ref", "HEAD")


# ---------- read-only (no approval needed) ----------

@mcp.tool()
def git_status() -> str:
    """Show git branch and changed files."""
    return clip(git("status", "--short", "--branch"))


@mcp.tool()
def git_diff() -> str:
    """Show uncommitted changes."""
    return clip(git("diff"))


# ---------- these change things (agent.py gates them behind approval) ----------

@mcp.tool()
def run_command(command: str) -> str:
    """Run a test or script in the project folder, e.g. 'pytest -x -q' or 'python main.py'."""
    try:
        r = subprocess.run(shlex.split(command), cwd=ROOT, capture_output=True, text=True, timeout=60)
        return clip(f"exit code {r.returncode}\n{r.stdout}\n{r.stderr}")
    except Exception as e:
        return f"Error running command: {e}"


@mcp.tool()
def edit_file(path: str, old_text: str, new_text: str) -> str:
    """Replace old_text with new_text in a file. old_text must appear exactly once."""
    try:
        full = safe_path(path)
        with open(full, "r") as f:
            content = f.read()
        if content.count(old_text) != 1:
            return f"Error: old_text found {content.count(old_text)} times, need exactly 1. Read the file and be more specific."
        with open(full, "w") as f:
            f.write(content.replace(old_text, new_text))
        return f"Edited {path}"
    except Exception as e:
        return f"Error editing {path}: {e}"


@mcp.tool()
def git_create_branch(name: str) -> str:
    """Create and switch to a new branch, e.g. 'fix/retrieval-threshold'."""
    if name in PROTECTED_BRANCHES:
        return "Refused: pick a new branch name."
    return git("checkout", "-b", name)


@mcp.tool()
def git_commit(message: str) -> str:
    """Stage all changes and commit them. Refuses on main/master."""
    if current_branch() in PROTECTED_BRANCHES:
        return "Refused: create a fix branch first."
    git("add", "-A")
    return git("commit", "-m", message)


@mcp.tool()
def git_push() -> str:
    """Push the current branch to origin. Refuses on main/master."""
    branch = current_branch()
    if branch in PROTECTED_BRANCHES:
        return "Refused: never push directly to main/master."
    return git("push", "-u", "origin", branch)


@mcp.tool()
def create_pull_request(title: str, body: str) -> str:
    """Open a GitHub pull request for the current branch (uses the gh CLI)."""
    r = subprocess.run(["gh", "pr", "create", "--title", title, "--body", body],
                       cwd=ROOT, capture_output=True, text=True, timeout=60)
    return clip(r.stdout + r.stderr)

if __name__ == "__main__":
    mcp.run()
